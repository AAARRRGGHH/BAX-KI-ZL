if(SMARTMIDI_SIMULATOR)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  add_subdirectory(vendor/SDL EXCLUDE_FROM_ALL)
  set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
  set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
  set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
  set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
  add_subdirectory(vendor/lvgl EXCLUDE_FROM_ALL)
  # LVGL 9.2.2 adds DLL attributes unconditionally on MSVC, even for static builds.
  if(MSVC)
    foreach(prop COMPILE_DEFINITIONS INTERFACE_COMPILE_DEFINITIONS)
      get_target_property(defs lvgl ${prop})
      list(FILTER defs EXCLUDE REGEX "^LV_ATTRIBUTE_EXTERN_DATA=")
      set_property(TARGET lvgl PROPERTY ${prop} "${defs}")
    endforeach()
  endif()
  file(GLOB UI_FONT_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/ui/fonts/*.c")
  add_library(controller_ui ui/ui.cpp ui/setup.cpp ui/power.cpp ui/boot_logo.cpp ui/travel.cpp ui/arpeggiator.cpp ui/chord_editor.cpp ui/charge_screen.cpp ui/bounce.cpp ui/bounce_mode.cpp experimental/bounce/menu.cpp experimental/bounce/view.cpp ${UI_FONT_SOURCES})
  target_include_directories(controller_ui PUBLIC include)
  target_link_libraries(controller_ui PUBLIC music_core bounce_core lvgl)
  add_executable(SmartMidiSimulator platform/desktop/main.cpp platform/desktop/device_mask.cpp)
  target_link_libraries(SmartMidiSimulator PRIVATE controller_ui SDL2::SDL2-static)
  target_compile_definitions(SmartMidiSimulator PRIVATE SDL_MAIN_HANDLED)
  if(WIN32)
    target_link_libraries(SmartMidiSimulator PRIVATE windowscodecs ole32)
  endif()
  add_custom_command(TARGET SmartMidiSimulator POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:SmartMidiSimulator>/assets"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/assets/simulator_mask.png" "$<TARGET_FILE_DIR:SmartMidiSimulator>/assets/simulator_mask.png")
  add_test(NAME simulator_interaction COMMAND SmartMidiSimulator --smoke-test)
  add_test(NAME companion_controls COMMAND SmartMidiSimulator --companion-controls-test)
  set_tests_properties(companion_controls PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy" TIMEOUT 20)
  add_executable(BounceSimulator experimental/bounce/main.cpp platform/desktop/device_mask.cpp)
  target_include_directories(BounceSimulator PRIVATE platform/desktop)
  target_compile_definitions(BounceSimulator PRIVATE SDL_MAIN_HANDLED)
  target_link_libraries(BounceSimulator PRIVATE bounce_core controller_ui SDL2::SDL2-static)
  if(WIN32)
    target_link_libraries(BounceSimulator PRIVATE windowscodecs ole32)
  endif()
  add_custom_command(TARGET BounceSimulator POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:BounceSimulator>/assets"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/assets/simulator_mask.png" "$<TARGET_FILE_DIR:BounceSimulator>/assets/simulator_mask.png")
  add_test(NAME bounce_simulator COMMAND BounceSimulator --smoke-test)
  set_tests_properties(bounce_simulator PROPERTIES TIMEOUT 20)
  add_executable(bounce_menu_tests tests/bounce_menu_tests.cpp)
  target_link_libraries(bounce_menu_tests PRIVATE bounce_core controller_ui)
  add_test(NAME bounce_menu COMMAND bounce_menu_tests)
  add_executable(bounce_integration_tests tests/bounce_integration_tests.cpp)
  target_link_libraries(bounce_integration_tests PRIVATE controller_ui)
  add_test(NAME bounce_integration COMMAND bounce_integration_tests)
  set_tests_properties(bounce_integration PROPERTIES TIMEOUT 30)
  add_executable(ui_power_confirm_tests tests/ui_power_confirm_tests.cpp)
  target_link_libraries(ui_power_confirm_tests PRIVATE controller_ui)
  add_test(NAME ui_power_confirmation COMMAND ui_power_confirm_tests)
  set_tests_properties(ui_power_confirmation PROPERTIES TIMEOUT 30)
